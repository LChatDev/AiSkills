# Пример политики, а не готовый конфиг конкретного сервера.
# Только для публичных статических ресурсов с версией в имени:
Cache-Control: public, max-age=31536000, immutable

# Для персональных или чувствительных ответов:
Cache-Control: private, no-store

# Для HTML, который должен проверяться при каждом использовании:
Cache-Control: no-cache
